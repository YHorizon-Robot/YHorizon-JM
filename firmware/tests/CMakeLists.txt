cmake_minimum_required(VERSION 3.22)
project(jm1_host_tests C)
enable_testing()
add_executable(dm4310_test dm4310_test.c ../App/compat/dm4310.c ../App/compat/comm_dm4310.c)
target_compile_features(dm4310_test PRIVATE c_std_11)
target_compile_definitions(dm4310_test PRIVATE CFG_DM4310=1 CFG_CAN_COMPAT=1)
target_include_directories(dm4310_test PRIVATE ../App ../App/compat ../Boards/YH_JM ../Hw/include ../Foc)
if(NOT MSVC)
    target_link_libraries(dm4310_test PRIVATE m)
endif()
add_test(NAME dm4310_session_and_codec COMMAND dm4310_test)

include(../cmake/can-profile.cmake)
foreach(model rs05 cybergear ak60-6 rmd-x lk-mf hightorque c620 c610)
    set(target "${model}_test")
    add_executable(${target} profile_test.c)
    jm1_add_profile(${target} "${model}")
    target_compile_features(${target} PRIVATE c_std_11)
    target_include_directories(${target} PRIVATE ../App ../App/compat ../Boards/YH_JM ../Hw/include ../Foc)
    if(NOT MSVC)
        target_link_libraries(${target} PRIVATE m)
    endif()
    add_test(NAME "${model}_session_and_codec" COMMAND ${target})
endforeach()

foreach(compat 0 1)
    set(target "commission_profile_${compat}_test")
    add_executable(${target} commission_profile_test.c ../App/cali_nv.c ../App/cali.c
        ../App/commission.c ../App/servo.c ../Foc/foc_current.c ../Foc/foc_encoder.c
        ../Foc/foc_math.c ../Foc/foc_svpwm.c ../Foc/foc_vernier.c)
    target_compile_features(${target} PRIVATE c_std_11)
    if(compat EQUAL 0)
        target_sources(${target} PRIVATE ../App/comm.c)
    endif()
    target_compile_definitions(${target} PRIVATE CFG_COMMISSION=1 CFG_CAN_COMPAT=${compat})
    target_include_directories(${target} PRIVATE ../App ../Boards/YH_JM ../Hw/include ../Foc)
    if(NOT MSVC)
        target_link_libraries(${target} PRIVATE m)
    endif()
    add_test(NAME ${target} COMMAND ${target})
endforeach()
